block    ::= stmt
           | '{' stmt* '}'
stmt     ::= ( expr | 'do' block 'while' '(' expr ')' ) ';'
           | ( ( 'if' | 'while' ) '(' expr ')' )? block
expr     ::= ID
           | CONST
           | expr ( '+' | '-' | '*' ) expr
_        ::= WS
          /* ws: definition */

<?TOKENS?>

ID       ::= [a-zA-Z] [a-zA-Z0-9]*
CONST    ::= [0-9]+
WS       ::= [ #x9#xd#xa]+
